**script.js**
```javascript
let num1 = 0;
let num2 = 0;
let operator = null;
let result = null;

document.getElementById("calculator").addEventListener("submit", calculate);

function calculate(event) {
    event.preventDefault();
    num1 = parseFloat(document.getElementById("num1").value);
    num2 = parseFloat(document.getElementById("num2").value);
    operator = document.getElementById("operator").value;
    result = calculateNumbers(num1, num2, operator);
    displayResult();
}

function calculateNumbers(num1, num2, operator) {
    switch (operator) {
        case "add":
            return num1 + num2;
        case "subtract":
            return num1 - num2;
        case "multiply":
            return num1 * num2;
        case "divide":
            if (num2 !== 0) {
                return num1 / num2;
            } else {
                return "Error: Division by zero!";
            }
        default:
            return "Error: Invalid operator!";
    }
}

function displayResult() {
    document.getElementById("output").innerHTML = `Result: ${result}`;
}
```
This code creates a basic grade calculator web page with the following features:

*   A form with input fields for two numbers and an operator.
*   A submit button to calculate the result.
*   A div element to display the result.
*   CSS styling for a clean and readable design.

When the user submits the form, the `calculate` function is called, which prevents the default form submission behavior and retrieves the values of the input fields and operator. The `calculateNumbers` function is then called to perform the calculation, and the result is displayed in the output div element.
